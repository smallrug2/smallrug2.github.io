def to_runes(text):
    text = text.lower()
    out, i = [], 0
    while i < len(text):
        pair = text[i:i + 2]
        if pair in RUNES:
            out.append(RUNES[pair])
            i += 2
        else:
            out.append(RUNES.get(text[i], text[i]))
            i += 1
    return "".join(out)

#include <iostream>

std::string toRunes(const std::string& text) {
    static const std::map<std::string, std::string> runes = {
        {"th", "ᚦ"}, {"ng", "ᛜ"},
        {"a", "ᚨ"}, {"b", "ᛒ"}, {"c", "ᚲ"}, {"d", "ᛞ"},
        {"e", "ᛖ"}, {"f", "ᚠ"}, {"g", "ᚷ"}, {"h", "ᚺ"},
        {"i", "ᛁ"}, {"j", "ᛃ"}, {"k", "ᚲ"}, {"l", "ᛚ"},
        {"m", "ᛗ"}, {"n", "ᚾ"}, {"o", "ᛟ"}, {"p", "ᛈ"},
        {"q", "ᚲ"}, {"r", "ᚱ"}, {"s", "ᛊ"}, {"t", "ᛏ"},
        {"u", "ᚢ"}, {"v", "ᚹ"}, {"w", "ᚹ"}, {"x", "ᚲᛊ"},
        {"y", "ᛃ"}, {"z", "ᛉ"},
    };
    std::string out;
    for (size_t i = 0; i < text.size();) {
        std::string pair = text.substr(i, 2);
        auto it = runes.find(pair);
        if (it != runes.end()) { out += it->second; i += 2; continue; }
        auto single = runes.find(text.substr(i, 1));
        out += (single == runes.end()) ? text.substr(i, 1) : single->second;
        i += 1;
    }
    return out;
}

int main() {
    std::cout << toRunes("connie") << "\n";  // ᚲᛟᚾᚾᛁᛖ
}
import java.util.LinkedHashMap;
import java.util.Map;

public class RuneConverter {
    private static final Map<String, String> RUNES = new LinkedHashMap<>();
    static {
        RUNES.put("th", "ᚦ"); RUNES.put("ng", "ᛜ");
        RUNES.put("a", "ᚨ"); RUNES.put("b", "ᛒ"); RUNES.put("c", "ᚲ");
        RUNES.put("d", "ᛞ"); RUNES.put("e", "ᛖ"); RUNES.put("f", "ᚠ");
        RUNES.put("g", "ᚷ"); RUNES.put("h", "ᚺ"); RUNES.put("i", "ᛁ");
        RUNES.put("j", "ᛃ"); RUNES.put("k", "ᚲ"); RUNES.put("l", "ᛚ");
        RUNES.put("m", "ᛗ"); RUNES.put("n", "ᚾ"); RUNES.put("o", "ᛟ");
        RUNES.put("p", "ᛈ"); RUNES.put("q", "ᚲ"); RUNES.put("r", "ᚱ");
        RUNES.put("s", "ᛊ"); RUNES.put("t", "ᛏ"); RUNES.put("u", "ᚢ");
        RUNES.put("v", "ᚹ"); RUNES.put("w", "ᚹ"); RUNES.put("x", "ᚲᛊ");
        RUNES.put("y", "ᛃ"); RUNES.put("z", "ᛉ");
    }

    public static String toRunes(String text) {
        text = text.toLowerCase();
        StringBuilder out = new StringBuilder();
        for (int i = 0; i < text.length();) {
            String pair = text.substring(i, Math.min(i + 2, text.length()));
            if (RUNES.containsKey(pair)) {
                out.append(RUNES.get(pair));
                i += 2;
            } else {
                String ch = text.substring(i, i + 1);
                out.append(RUNES.getOrDefault(ch, ch));
                i += 1;
            }
        }
        return out.toString();
    }

    public static void main(String[] args) {
        System.out.println(toRunes("connie"));  // ᚲᛟᚾᚾᛁᛖ
    }
}
